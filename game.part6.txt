  function projectileThreat(enemy,projectile,guided=false){
    if(projectile.life<=0)return null;
    const offset=enemy.mesh.position.clone().sub(projectile.mesh.position);
    if(offset.lengthSq()>(guided?44:65)**2)return null;
    const flight=guided?projectile.direction.clone().multiplyScalar(projectile.speed):projectile.velocity.clone();
    const relative=flight.clone().sub(enemy.velocity),speedSq=relative.lengthSq();if(speedSq<1)return null;
    const eta=offset.dot(relative)/speedSq;if(eta<=.015||eta>(guided?1:.5))return null;
    const miss=offset.clone().addScaledVector(relative,-eta).length();
    if(miss>enemy.radius+(guided?3:1.1)&&!(guided&&projectile.target===enemy))return null;
    return {projectile,guided,eta,direction:flight.normalize()};
  }

    // Flash first; rising smoke and spinning fragments outlive the fireball.
    const density=effectDensity();
    particle(position,0xffe7b5,3.2*power,.18,V(),'flash',true);
    particle(position,0xffca7a,1.0*power,.6,V(),'ring',true);
    for(let i=0;i<Math.max(2,Math.round(4*density));i++){
      const offset=V(rand(-.6,.6),rand(-.3,.6),rand(-.6,.6)).multiplyScalar(power);
      particle(position.clone().add(offset),i%2?0xff711d:0xffb849,rand(1.3,2.2)*power,rand(.32,.65),offset.multiplyScalar(3).add(V(0,1.8,0)),'fire',true);
    }
    burst(position,0xffc36c,Math.round(18*power));
    for(let i=0;i<Math.max(2,Math.round(5*density*power));i++)particle(position,0x4b514f,rand(.17,.34)*power,rand(1.2,2.4),V(rand(-7,7),rand(3,10),rand(-7,7)).multiplyScalar(power),'debris');
    for(let i=0;i<Math.max(3,Math.round(9*density));i++)particle(position.clone().add(V(rand(-.8,.8),rand(0,.5),rand(-.8,.8))),i%2?0x3f423e:0x6c6960,rand(1.2,2.1)*power,rand(1.4,2.7),V(rand(-1.6,1.6),rand(1.8,4.5),rand(-1.6,1.6)),'smoke');
    const volume=clamp(1-position.distanceTo(player.position)/160,.16,1);
    noiseSound('blast',volume*Math.min(1.2,power));sound(78,.3,.06*volume);kickCamera(position,.8*power);
  }

  function updateEffects(dt){
    cameraImpulse=Math.max(0,cameraImpulse-dt*2.8);
    while(particles.length>effectLimit())removeParticle(particles.shift());
    for(const p of particles){
      p.life-=dt;const remain=clamp(p.life/p.maxLife,0,1),age=1-remain;
      if(p.kind==='spark'||p.kind==='debris')p.velocity.y-=9.8*dt;
      else if(p.kind==='smoke'){p.velocity.multiplyScalar(Math.exp(-.45*dt));p.velocity.y+=.7*dt;}
      p.mesh.position.addScaledVector(p.velocity,dt);
      if(p.kind==='debris'){
        p.mesh.rotation.x+=p.spin*dt*2;p.mesh.rotation.z+=p.spin*dt;
        if(p.mesh.position.y< -7.35){p.mesh.position.y=-7.35;p.velocity.y=Math.abs(p.velocity.y)*.22;p.velocity.x*=.7;p.velocity.z*=.7;}
        p.mesh.material.opacity=Math.min(1,remain*3);
      }else if(p.kind==='smoke'){
        p.mesh.scale.setScalar(p.size*(1+age*2.8));p.mesh.material.opacity=Math.min(1,age*9)*remain*.58;p.mesh.material.rotation+=p.spin*dt*.12;
      }else if(p.kind==='ring'){
        p.mesh.scale.setScalar(p.size*(1+age*8));p.mesh.material.opacity=remain*remain*.4;
      }else if(p.kind==='flash'||p.kind==='fire'){
        p.mesh.scale.setScalar(p.size*(1+age*(p.kind==='flash'?1.1:1.8)));p.mesh.material.opacity=remain*remain*(p.kind==='flash'?.9:.85);
      }else{
        p.mesh.material.opacity=remain;p.mesh.scale.setScalar(p.size*(.4+remain*.6));
        if(p.velocity.lengthSq()>.01)p.mesh.quaternion.setFromUnitVectors(V(0,0,1),p.velocity.clone().normalize());
      }
    }
    particles=particles.filter(p=>{if(p.life>0)return true;removeParticle(p);return false;});
    foes=foes.filter(e=>{if(!e.dead)return true;disposeObject(e.mesh);return false;});
  }
  function renderWorld(){
    // Shake only the rendered view: aiming, missile lock and flight use the steady camera.
    if(!running||cameraImpulse<=0||!profile.settings.cameraShake||profile.settings.effects==='low'||reducedEffectsMotion()){renderer.render(scene,camera);return;}
    const position=camera.position.clone(),quaternion=camera.quaternion.clone(),amount=cameraImpulse*(profile.settings.camera==='fpv'?.45:1);
    try{
      camera.translateX(Math.sin(clock*49)*amount*.075);camera.translateY(Math.sin(clock*61)*amount*.05);
      camera.rotateZ(Math.sin(clock*43)*amount*.004);camera.updateMatrixWorld(true);renderer.render(scene,camera);
    }finally{camera.position.copy(position);camera.quaternion.copy(quaternion);camera.updateMatrixWorld(true);}
  }

  function detonate(missile) {
    if (missile.life <= 0||!running) return;
    missile.life = 0; const position = missile.mesh.position;
    explosion(position,1.4);
    const affected=foes.filter(e=>!e.dead&&e.mesh.position.distanceTo(position)<7.5+e.radius*.3&&!lineBlocked(position,e.mesh.position));
    if(affected.length)combatLog.missileHits++;
    for(const e of affected){const damage=Math.min(Math.max(0,e.hp),flightStats().missile);combatLog.damageDealt+=damage;combatLog.lastHit=damage;e.hp-=damage;e.hitReaction=.22;hitTime=.18;}
    for(const e of affected)if(e.hp<=0)destroyEnemy(e);
  }
  function updateMissiles(dt) {
    for (const m of missiles) {
      if(!running)break;
      if (m.target && !m.target.dead) {
        const desired = m.target.mesh.position.clone().sub(m.mesh.position).normalize();
        m.direction.lerp(desired, 1 - Math.exp(-7 * dt)).normalize();
      }
      m.speed = Math.min(74, m.speed + 32 * dt);
      const previous = m.mesh.position.clone();
      m.mesh.position.addScaledVector(m.direction, m.speed * dt); m.life -= dt;
      m.mesh.quaternion.setFromUnitVectors(FORWARD, m.direction);
      m.mesh.userData.flame.scale.y = .85 + Math.random() * .25;
      m.smoke -= dt;
      if (m.smoke <= 0) { m.smoke=touchActive?.08:.05;particle(m.mesh.position,0xc7c7bc,.45,.85,V(0,.9,0),'smoke'); }
      const wall=firstObstacle(previous,m.mesh.position,.2);if(wall){m.mesh.position.copy(previous).lerp(m.mesh.position,Math.max(0,wall.t-.01));detonate(m);}
      for (const e of foes) if (!e.dead && segmentDistanceSq(previous, m.mesh.position, e.mesh.position) < (e.radius+.9)**2) { detonate(m); break; }
      if (m.mesh.position.y < -7) m.life = 0;
    }
    missiles = missiles.filter(m => { if (m.life > 0) return true; disposeObject(m.mesh); return false; });
  }
