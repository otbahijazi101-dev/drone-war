  function fleetAction(action){
    if(action==='fleet'){garageModel=profile.selectedDrone;garageMessage='';showPanel('fleet');return true;}
    if(action==='fleet-all'||action==='fleet-owned'){fleetOwned=action==='fleet-owned';renderPanel();return true;}
    if(action.startsWith('class-')){const cls=action.slice(6);if(['all','D','C','B','A','S'].includes(cls))fleetFilter=cls;renderPanel();return true;}
    const [op,...parts]=action.split('-'),value=parts.join('-');
    if(op==='inspect'){if(DRONES.some(d=>d.id===value)){garageModel=value;garageMessage='';renderPanel();}return true;}
    if(op==='purchase'){buyDrone(value);return true;}
    if(op==='equip'){equipDrone(value);return true;}
    if(op==='customize'){if(ownsDrone(value)){garageModel=value;garageMessage='';showPanel('customize');}return true;}
    if(op==='finish'){if(Object.hasOwn(FINISHES,value))changePaint('finish',value);return true;}
    if(op==='paint'&&value==='reset'){if(ownsDrone(garageModel)){profile.paintJobs[garageModel]=defaultPaint(garageModel);garageMessage='Fabrika görünümü uygulandı.';saveProfile();renderPanel();}return true;}
    if(op==='paint'){const c=PAINT_COLORS[Number(value)];if(c)changePaint('color',c[0]);return true;}
    if(op==='led'){const c=LED_COLORS[Number(value)];if(c)changePaint('led',c[0]);return true;}
    if(op==='blades'){if(value==='2'||value==='3')changePaint('blades',Number(value));return true;}
    if(action==='preview-reset'){previewAngle=.65;previewTilt=.30;previewDistance=8.8;previewIdle=0;return true;}
    return false;
  }

  function framePreviewCamera(width,height,angle=previewAngle,tilt=previewTilt,distance=previewDistance){
    garageCamera.aspect=width/height;garageCamera.updateProjectionMatrix();
    const vertical=T.MathUtils.degToRad(garageCamera.fov/2),halfAngle=Math.min(vertical,Math.atan(Math.tan(vertical)*width/height));

  function storeMissionReport(win){
    const r=sessionReport=captureReport(win?'success':'defeat');if(r.practice)return;
    const key=recordKey(r.map,r.missionId,r.difficulty),old=profile.records[key];
    const rec=old?{...old}:{attempts:0,wins:0,bestScore:0,bestTime:null,bestAccuracy:null,leastDamage:null};
    rec.attempts++;
    if(win){
      rec.wins++;
      if(!old?.wins)r.medals.push('İLK TAMAMLAMA');
      else{
        if(r.score>rec.bestScore)r.medals.push('PUAN REKORU');
        if(rec.bestTime===null||r.time<rec.bestTime-.01)r.medals.push('SÜRE REKORU');
        if(r.accuracy!==null&&(rec.bestAccuracy===null||r.accuracy>rec.bestAccuracy+.01))r.medals.push('İSABET REKORU');
        if(rec.leastDamage===null||r.damageTaken<rec.leastDamage)r.medals.push('EN AZ HASAR');
      }
      rec.bestScore=Math.max(rec.bestScore,r.score);rec.bestTime=rec.bestTime===null?r.time:Math.min(rec.bestTime,r.time);
      if(r.accuracy!==null)rec.bestAccuracy=rec.bestAccuracy===null?r.accuracy:Math.max(rec.bestAccuracy,r.accuracy);
      rec.leastDamage=rec.leastDamage===null?r.damageTaken:Math.min(rec.leastDamage,r.damageTaken);
    }
    profile.records[key]=rec;r.record={...rec};saveProfile();
  }
