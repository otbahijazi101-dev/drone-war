  }
  async function readBackup(file){
    pendingBackup=null;if(!file)return;
    try{
      if(file.size>262144)throw new Error('Boyut');const data=JSON.parse(await file.text());
      if(data?.format!=='drone-arena-save'||data.version!==1||!data.profile||typeof data.profile!=='object'||!Number.isFinite(data.profile.credits)||!Number.isFinite(data.profile.xp)||!Array.isArray(data.profile.ownedDrones))throw new Error('Biçim');
      pendingBackup=cleanProfile(data.profile);backupMessage='Dosya kontrol edildi. Aşağıdan yüklemeyi tamamlayabilirsin.';
    }catch(_){backupMessage='Bu dosya geçerli bir Drone Arena yedeği değil. İlerlemen değiştirilmedi.';}
    if(activeView==='backup')renderPanel();
  }
  function useBackup(value){
    running=false;practiceActive=false;mission=null;profile=cleanProfile(value);selectedMission=profile.unlockedMission;sessionDifficulty=profile.settings.difficulty;
    thumbnailCache.clear();applyMap(profile.selectedMap);reset();applyQuality();saveProfile();pendingBackup=null;backupMessage='Kayıt yüklendi. Uçuşa devam edebilirsin.';showPanel('backup');
  }
  function flightFeatureAction(a){
    if(a==='start-chase'){startChase();return true;}
    if(a.startsWith('difficulty-')){const id=a.slice(11);if(DIFFICULTIES[id]){profile.settings.difficulty=id;saveProfile();renderPanel()}return true;}
    if(a==='camera'){toggleCamera();if(!running)renderPanel();return true;}
    if(a==='backup'){backupMessage='';pendingBackup=null;showPanel('backup');return true;}
    if(a==='export-save'){exportSave();return true;}
    if(a==='cancel-backup'){pendingBackup=null;renderPanel();return true;}
    if(a==='apply-backup'){if(pendingBackup){backupUndo=JSON.parse(JSON.stringify(profile));useBackup(pendingBackup)}return true;}
    if(a==='undo-backup'){if(backupUndo){const old=backupUndo;backupUndo=null;useBackup(old);backupMessage='Önceki ilerleme geri yüklendi.';renderPanel()}return true;}
    return false;
  }
