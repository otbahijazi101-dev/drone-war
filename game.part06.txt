  function storeMissionReport(win){
    const r=sessionReport=captureReport(win?'success':'defeat');if(r.practice)return;
    const key=recordKey(r.map,r.missionId,r.difficulty),old=profile.records[key];
    const rec=old?{...old}:{attempts:0,wins:0,bestScore:0,bestTime:null,bestAccuracy:null,leastDamage:null};
    rec.attempts++;
    if(win){
      rec.wins++;
      if(!old?.wins)r.medals.push('İLK TAMAMLAMA');
      else{
        if(r.score>rec.bestScore)r.medals.push('PUAN REKORU');
        if(rec.bestTime===null||r.time<rec.bestTime-.01)r.medals.push('SÜRE REKORU');
        if(r.accuracy!==null&&(rec.bestAccuracy===null||r.accuracy>rec.bestAccuracy+.01))r.medals.push('İSABET REKORU');
        if(rec.leastDamage===null||r.damageTaken<rec.leastDamage)r.medals.push('EN AZ HASAR');
      }
      rec.bestScore=Math.max(rec.bestScore,r.score);rec.bestTime=rec.bestTime===null?r.time:Math.min(rec.bestTime,r.time);
      if(r.accuracy!==null)rec.bestAccuracy=rec.bestAccuracy===null?r.accuracy:Math.max(rec.bestAccuracy,r.accuracy);
      rec.leastDamage=rec.leastDamage===null?r.damageTaken:Math.min(rec.leastDamage,r.damageTaken);
    }
    profile.records[key]=rec;r.record={...rec};saveProfile();
  }
