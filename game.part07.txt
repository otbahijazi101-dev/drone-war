  function sound(freq=420,duration=.07,volume=.028){
    if(!profile.settings.sound||!audioContext)return;
    try{
      const now=audioContext.currentTime,o=audioContext.createOscillator(),g=audioContext.createGain();
      o.type='triangle';o.frequency.setValueAtTime(freq,now);o.frequency.exponentialRampToValueAtTime(Math.max(60,freq*.55),now+duration);
      g.gain.setValueAtTime(volume,now);g.gain.exponentialRampToValueAtTime(.0001,now+duration);
