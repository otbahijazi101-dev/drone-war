  function spawnEnemy() {
    if(practiceActive)return spawnPracticeTarget();
    if(!mission||mission.kind==='race'||(['combat','boss'].includes(mission.kind)&&missionKills>=mission.goal))return false;
    const bossPhase=mission.boss && missionKills>=mission.goal-1;
    if(endlessActive){const regularCap=mission.boss?mission.goal-1:mission.goal;if(!bossPhase&&endlessSpawned>=regularCap)return false;if(bossPhase&&bossSpawned)return false;}
    if(bossPhase&&bossSpawned)return false;
    if(foes.filter(e=>!e.dead).length>= (bossPhase?1:enemyLimit()))return false;
    let type='striker';const roll=Math.random();
    if(bossPhase){type='boss';bossSpawned=true;}
    else if(mission.id>=3&&roll<Math.min(.3,.08+mission.id*.02))type='armored';
    else if(roll<.48)type='scout';
    const angle=yaw+(isChase()?Math.PI:0)+rand(-1.3,1.3);
    const radius=isChase()?rand(85,125):type==='boss'?57:rand(34,58);
    const pos=player.position.clone().add(V(-Math.sin(angle)*radius,rand(-3,4),-Math.cos(angle)*radius));
    const spawnBoundary=isChase()?535:88;pos.x=clamp(pos.x,-spawnBoundary,spawnBoundary);pos.z=clamp(pos.z,-spawnBoundary,spawnBoundary);pos.y=clamp(pos.y,4,flightCeiling()-5);
    if(pos.distanceTo(player.position)<22){
      const inward=V(-player.position.x,0,-player.position.z).normalize().multiplyScalar(42);
      pos.copy(player.position).add(inward);pos.y=clamp(player.position.y+3,4,28);
    }
    const radiusHit=type==='boss'?3.6:type==='armored'?2.45:1.8;
    let hitpoints=type==='boss'?20+mission.id*2:type==='armored'?7+Math.floor(mission.id/3):
      type==='scout'?2+Math.floor(mission.id/4):4+Math.floor(mission.id/4);
    if(endlessActive)hitpoints=Math.max(2,Math.round(hitpoints*endlessHealthScale(endlessLevel)*(type==='boss'?1.45:1)));
    for(const obstacle of activeColliders)if(obstacle.clone().expandByScalar(4).containsPoint(pos))pos.y=Math.min(flightCeiling()-3,obstacle.max.y+7);
    const enemy={lastSeen:player.position.clone(),unseen:0,mesh:makeDrone(true,type),hp:hitpoints,maxHP:hitpoints,radius:radiusHit,type,dead:false,
      speed:(type==='boss'?4.8:type==='armored'?5.2:type==='scout'?10:7.2)*(endlessActive?endlessSpeedScale(endlessLevel):1),
      velocity:V(),fire:rand(type==='boss'?2.9:4.5,6.5)*(endlessActive?endlessFireScale(endlessLevel):1),phase:angle,orbit:Math.random()<.5?-1:1,
      state:'attack',stateTime:rand(2,5),evadeCooldown:0,evadeRemaining:0,evadeDirection:V(),missileTimer:rand(12,19),smokeTime:0,attacksAsset:!!missionAsset&&Math.random()<.42};
    enemy.mesh.scale.setScalar(type==='boss'?(endlessActive?2.05:1.85):type==='armored'?1.22:type==='scout'?.87:1);
    enemy.mesh.position.copy(pos);scene.add(enemy.mesh);foes.push(enemy);if(endlessActive)endlessSpawned++;return true;
  }
